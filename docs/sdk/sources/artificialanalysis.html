sdk.sources.artificialanalysis

ArtificialAnalysis.ai integration.

Data flow (replicated from the site's client-side JS):

  1. The model leaderboard page embeds a manifest in its RSC payload: {"path": "/data/<hash>.txt", "key": "<64-hex AES key>"}.
  2. Fetch the data file - it is a binary blob: AES-GCM(plaintext gzip json), where the passphrase is the manifest key, the raw key bytes are the AES key, and the IV is sha256(key)[:12].
  3. Decrypt, gunzip, JSON.parse -> the full model dataset.

The endpoint returns ~900 models with intelligence/agentic/omniscience indices, per-benchmark scores, pricing variants (blended ratios, cache, images), and performance distributions by prompt type.

351# The site embeds a manifest in its RSC flight payload. The raw HTML keeps
352# the flight escaped ("\\" before every quote) AND the payload is split across
353# multiple script tags, so detection has to handle several shapes:
354#   - {"manifest":{"path":"/data/<hex>.txt","key":"<64-hex>"}}
355#   - the escaped form  {\\"manifest\\":{\\"path\\":\\"...\\",\\"key\\":\\"...\\"}}
356#   - key order swapped, extra fields, unicode escapes, line breaks
357_MANIFEST_PATH_RE = re.compile(
358    r'"(?:path|file|url)"\s*:\s*"(/data/[a-f0-9]{16,64}\.txt)"'
359)
360_MANIFEST_KEY_RE = re.compile(
361    r'"(?:key|decryptKey|encryptionKey)"\s*:\s*"([a-f0-9]{64})"'
362)
363
364# escaped variant: every quote is preceded by a backslash inside the flight string
365_ESCAPED_PATH_RE = re.compile(
366    r'\\"(?:path|file|url)\\":\\"(/data/[a-f0-9]{16,64}\.txt)\\"'
367)
368_ESCAPED_KEY_RE = re.compile(
369    r'\\"(?:key|decryptKey|encryptionKey)\\":\\"([a-f0-9]{64})\\"'
370)
371
372
373def _find_manifests_in_text(text: str) -> list[tuple[str, str]]:
374    """Extract (path, key) pairs from a payload, order preserved.
375
376    Tries both plain and escaped regexes so key order or extra fields do
377    not matter. Only pairs where both halves match are returned.
378    """
379    out: list[tuple[str, str]] = []
380    for path_re, key_re in ((_MANIFEST_PATH_RE, _MANIFEST_KEY_RE), (_ESCAPED_PATH_RE, _ESCAPED_KEY_RE)):
381        paths = path_re.findall(text)
382        keys = key_re.findall(text)
383        n = max(len(paths), len(keys))
384        for i in range(n):
385            p = paths[i] if i < len(paths) else None
386            k = keys[i] if i < len(keys) else None
387            if p and k:
388                out.append((p, k))
389    return out

392def fetch_aa_manifests(client: OpenRouterClient, page: str = "/models") -> list[tuple[str, str]]:
393    """Fetch an AA page and pull every data manifest out of its payload.
394
395    Returns a list of (path, key) pairs in page order. Handles both the
396    escaped flight form and plain JSON by scanning every script tag body
397    (not just __next_f.push) and the unescaped HTML as fallback.
398    """
399    body = client.get(AA_BASE + page, timeout=60).text
400
401    manifests: list[tuple[str, str]] = []
402    seen: set[tuple[str, str]] = set()
403
404    candidates: list[str] = []
405    # 1. every __next_f.push([N,"..."]) script: parse the array arg to raw string
406    for m in re.finditer(r"self\.__next_f\.push\((.*?)\)</script>", body, re.S):
407        candidates.append(m.group(1))
408    # 2. any script containing manifest keys (escaped or not)
409    for m in re.finditer(r"<script[^>]*>([^<]{200,}?)</script>", body, re.S):
410        candidates.append(m.group(1))
411    # 3. the whole HTML body itself (last resort)
412    candidates.append(body)
413
414    for raw in candidates:
415        payloads: list[str] = [raw]
416        stripped = raw.strip()
417        if stripped.startswith("[") and "]" in stripped:
418            try:
419                arr = json.loads(stripped)
420                if len(arr) > 1 and isinstance(arr[1], str):
421                    payloads.insert(0, arr[1])
422            except Exception:
423                pass
424        for payload in payloads:
425            for path, key in _find_manifests_in_text(payload):
426                if (path, key) not in seen:
427                    seen.add((path, key))
428                    manifests.append((path, key))
429
430    return manifests

433def _decrypt_dataset(client: OpenRouterClient, path: str, keyhex: str):
434    """Fetch + decrypt + gunzip + JSON.parse one manifest. Validates each step."""
435    if not re.fullmatch(r"[a-f0-9]{64}", keyhex):
436        raise ValueError(f"manifest key is not 64 hex chars: {keyhex[:12]}...")
437    blob = client.get(AA_BASE + path, timeout=60).content
438    if not blob:
439        raise ValueError(f"empty data blob for {path}")
440
441    key = bytes.fromhex(keyhex)
442    iv = hashlib.sha256(key).digest()[:12]
443    try:
444        plain = AESGCM(key).decrypt(iv, blob, None)
445    except Exception as e:
446        # fall back: some blobs may be plain gzip (no AES wrapping)
447        try:
448            data = json.loads(gzip.decompress(blob).decode("utf-8", errors="replace"))
449            return data.get("models", data) if isinstance(data, dict) else data
450        except Exception:
451            raise ValueError(

537def fetch_aa_initial_models(client: OpenRouterClient, page: str = "/models") -> list[AAModel]:
538    """Extract the flagship rows (initialModels) directly from the RSC flight.
539
540    These rows carry the full benchmark suite (intelligence/agentic indices,
541    HLE, GPQA, AIME25, scicode, ...) that the /data/*.txt manifest lacks.
542    """
543    body = client.get(AA_BASE + page, timeout=60).text
544    scripts = re.findall(r"self\.__next_f\.push\((.*?)\)</script>", body, re.S)
545    rows: list[AAModel] = []
546    for s_ in scripts:
547        try:
548            arr = json.loads(s_)
549            payload = arr[1] if len(arr) > 1 and isinstance(arr[1], str) else ""
550        except Exception:
551            continue
552        if "initialModels" not in payload:
553            continue
554        decoded = payload.encode("utf-8", errors="replace").decode("unicode_escape", errors="replace")
555        for line in decoded.split(chr(10)):
556            m = re.match(r"\d+:(.*)", line, re.S)
557            if not m:
558                continue
559            try:
560                val = json.loads(m.group(1))
561            except Exception:
562                continue
563            found = _walk_for_key(val, "initialModels")
564            if isinstance(found, list):
565                rows.extend(AAModel.from_raw(r) for r in found if isinstance(r, dict))
566    seen = set()
567    out = []
568    for m in rows:
569        k = m.model_slug or m.slug
570        if k and k not in seen:
571            seen.add(k)
572            out.append(m)
573    return out

577    """Merge flagship benchmark rows + broad manifest dataset by model slug.
578
579    Prefers the flagship row when it exists (benchmarks present); otherwise
580    keeps the manifest row (broad coverage). Per-source perf fields are kept
581    from whichever row carries them.
582    """

584    for m in fetch_aa_initial_models(client, page="/models"):
585        key = m.model_slug or m.slug
586        if key:
587            merged[key] = m
588    for m in fetch_aa_dataset(client, page="/models"):

590        if not key:
591            continue

393def fetch_aa_manifests(client: OpenRouterClient, page: str = "/models") -> list[tuple[str, str]]:
394    """Fetch an AA page and pull every data manifest out of its payload.
395
396    Returns a list of (path, key) pairs in page order. Handles both the
397    escaped flight form and plain JSON by scanning every script tag body
398    (not just __next_f.push) and the unescaped HTML as fallback.
399    """
400    body = client.get(AA_BASE + page, timeout=60).text
401
402    manifests: list[tuple[str, str]] = []
403    seen: set[tuple[str, str]] = set()
404
405    candidates: list[str] = []
406    # 1. every __next_f.push([N,"..."]) script: parse the array arg to raw string
407    for m in re.finditer(r"self\.__next_f\.push\((.*?)\)</script>", body, re.S):
408        candidates.append(m.group(1))
409    # 2. any script containing manifest keys (escaped or not)
410    for m in re.finditer(r"<script[^>]*>([^<]{200,}?)</script>", body, re.S):
411        candidates.append(m.group(1))
412    # 3. the whole HTML body itself (last resort)
413    candidates.append(body)
414
415    for raw in candidates:
416        payloads: list[str] = [raw]
417        stripped = raw.strip()
418        if stripped.startswith("[") and "]" in stripped:
419            try:
420                arr = json.loads(stripped)
421                if len(arr) > 1 and isinstance(arr[1], str):
422                    payloads.insert(0, arr[1])
423            except Exception:
424                pass
425        for payload in payloads:
426            for path, key in _find_manifests_in_text(payload):
427                if (path, key) not in seen:
428                    seen.add((path, key))
429                    manifests.append((path, key))
430
431    return manifests
